cmake_minimum_required(VERSION 3.25)
project(CTRRuntime LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CTR_SANITIZERS "Sanitize the host adapter (core is built separately)" OFF)
option(CTR_INCLUDE_PRIVATE_SKIN "Bundle locally supplied skin artwork (not for public releases)" OFF)
option(CTR_BUILD_FRONTEND "Build Bergamot application bundles" ON)
option(CTR_BUILD_HOST_TOOLS "Build Mac diagnostic executables" ON)
set(CTR_PLATFORM macos CACHE STRING "macos, ios or ios-simulator")
set(CTR_SIGNING_TEAM "" CACHE STRING "Optional Apple development team for device builds")
set(CORE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/build/${CTR_PLATFORM}/bin/Release/azahar_libretro.dylib")
add_library(azahar SHARED IMPORTED)
set_target_properties(azahar PROPERTIES IMPORTED_LOCATION "${CORE_PATH}")
if(CTR_PLATFORM STREQUAL "ios-simulator")
    set(MVK_SLICE ios-arm64_x86_64-simulator)
elseif(CTR_PLATFORM STREQUAL "ios")
    set(MVK_SLICE ios-arm64)
else()
    set(MVK_SLICE macos-arm64_x86_64)
endif()
set(MVK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/moltenvk-${CTR_PLATFORM}/MoltenVK/MoltenVK")
add_library(moltenvk STATIC IMPORTED)
set_target_properties(moltenvk PROPERTIES IMPORTED_LOCATION "${MVK_ROOT}/static/MoltenVK.xcframework/${MVK_SLICE}/libMoltenVK.a")
add_library(ctr-runtime STATIC src/runtime.cpp src/vulkan_backend.cpp)
target_include_directories(ctr-runtime PUBLIC src
    vendor/azahar/src
    vendor/azahar/externals/libretro-common/libretro-common/include
    vendor/azahar/externals/vulkan-headers/include)
# MoltenVK's configuration query guards queue-ordered presentation. The pinned
# packages share one version; the Simulator archive carries no headers.
set(MVK_INCLUDE "${MVK_ROOT}/include")
foreach(MVK_HEADER_PLATFORM macos ios)
    if(NOT EXISTS "${MVK_INCLUDE}/MoltenVK/mvk_private_api.h")
        set(MVK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/moltenvk-${MVK_HEADER_PLATFORM}/MoltenVK/MoltenVK/include")
    endif()
endforeach()
target_include_directories(ctr-runtime SYSTEM PUBLIC "${MVK_INCLUDE}")
target_compile_options(ctr-runtime PRIVATE -Wall -Wextra -Wpedantic)
add_library(ctr-metal-presenter STATIC src/metal_presenter.mm)
target_compile_options(ctr-metal-presenter PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
target_link_libraries(ctr-metal-presenter PUBLIC ctr-runtime)
target_link_libraries(ctr-runtime PUBLIC azahar moltenvk
    "-framework Foundation" "-framework Metal" "-framework QuartzCore" "-framework IOSurface" "-framework CoreGraphics")
if(NOT CTR_PLATFORM MATCHES "^ios")
    target_link_libraries(ctr-runtime PUBLIC "-framework AppKit" "-framework IOKit")
else()
    # MoltenVK references UIView/UIScreen even in an independent headless host.
    target_link_libraries(ctr-runtime PUBLIC "-framework UIKit")
endif()
if(CTR_BUILD_FRONTEND AND CTR_PLATFORM MATCHES "^ios")
    enable_language(OBJCXX)
    add_executable(CTRRuntime MACOSX_BUNDLE ios/main.mm)
    target_compile_options(CTRRuntime PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
    target_link_libraries(CTRRuntime PRIVATE ctr-metal-presenter "-framework UIKit" "-framework SceneKit" "-framework UniformTypeIdentifiers" "-framework AVFoundation")
    set_target_properties(CTRRuntime PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/ios/Info.plist"
        BUILD_RPATH "@executable_path/Frameworks"
        XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "local.skyfish.ctrruntime"
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
    if(CTR_PLATFORM STREQUAL "ios-simulator")
        set_target_properties(CTRRuntime PROPERTIES XCODE_ATTRIBUTE_SUPPORTED_PLATFORMS "iphonesimulator")
    endif()
    if(CTR_SIGNING_TEAM)
        set_target_properties(CTRRuntime PROPERTIES
            XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED YES
            XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
            XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${CTR_SIGNING_TEAM}"
            XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "local.skyfish.ctrruntime"
            XCODE_EMBED_FRAMEWORKS "${CORE_PATH}"
            XCODE_EMBED_FRAMEWORKS_CODE_SIGN_ON_COPY YES)
    else()
        add_custom_command(TARGET CTRRuntime POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:CTRRuntime>/Frameworks"
            COMMAND ${CMAKE_COMMAND} -E copy "${CORE_PATH}" "$<TARGET_BUNDLE_DIR:CTRRuntime>/Frameworks/azahar_libretro.dylib")
    endif()
    if(CTR_INCLUDE_PRIVATE_SKIN)
        file(GLOB CTR_SKIN_FILES "${CMAKE_CURRENT_SOURCE_DIR}/ios/Skins/WhiteN3DS/*")
        target_sources(CTRRuntime PRIVATE ${CTR_SKIN_FILES})
        set_source_files_properties(${CTR_SKIN_FILES} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/WhiteN3DS")
    endif()
    file(GLOB CTR_PIXEL_ART "${CMAKE_CURRENT_SOURCE_DIR}/ios/PixelArt/exports/*.png")
    list(FILTER CTR_PIXEL_ART EXCLUDE REGEX "-(icon-(gc|wii|wiiu|switch|ps1|ps2|psp)|control-(plus|minus|cross|circle|square|triangle|z|l1|l2|l3|r1|r2|r3))-[0-9]+\\.png$|-sheet\\.png$")
    target_sources(CTRRuntime PRIVATE ${CTR_PIXEL_ART})
    set_source_files_properties(${CTR_PIXEL_ART} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/PixelArt")
    if(CMAKE_GENERATOR STREQUAL "Xcode")
        target_sources(CTRRuntime PRIVATE artifacts/arm11-probe.3dsx NOTICE.txt LICENSE THIRD_PARTY_NOTICES.txt)
        set_source_files_properties(artifacts/arm11-probe.3dsx NOTICE.txt LICENSE THIRD_PARTY_NOTICES.txt PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    else()
        add_custom_command(TARGET CTRRuntime POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/artifacts/arm11-probe.3dsx" "$<TARGET_BUNDLE_DIR:CTRRuntime>/arm11-probe.3dsx")
    endif()
endif()
if(CTR_SANITIZERS)
    target_compile_options(ctr-runtime PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ctr-runtime PUBLIC -fsanitize=address,undefined)
endif()
if(CTR_BUILD_HOST_TOOLS AND NOT CTR_PLATFORM MATCHES "^ios")
    add_executable(ctr-probe src/probe_main.cpp)
    target_compile_options(ctr-probe PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ctr-probe PRIVATE ctr-runtime)
    set_target_properties(ctr-probe PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}/build/macos/bin/Release")
    add_executable(ctr-metal-probe src/metal_probe.mm)
    target_compile_options(ctr-metal-probe PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
    target_link_libraries(ctr-metal-probe PRIVATE ctr-metal-presenter)
    set_target_properties(ctr-metal-probe PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}/build/macos/bin/Release")
    add_executable(ctr-skin-presenter tests/skin_presenter.mm)
    target_compile_options(ctr-skin-presenter PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
    target_link_libraries(ctr-skin-presenter PRIVATE ctr-metal-presenter)
    add_executable(ctr-frame-pacer tests/frame_pacer.cpp)
    add_executable(ctr-display-layout tests/display_layout.cpp)
    target_link_libraries(ctr-display-layout PRIVATE ctr-runtime)
    target_compile_options(ctr-display-layout PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ctr-frame-pacer PRIVATE ctr-runtime)
    target_compile_options(ctr-frame-pacer PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(ctr-lifecycle tests/lifecycle.cpp)
    target_compile_options(ctr-lifecycle PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ctr-lifecycle PRIVATE ctr-runtime)
    set_target_properties(ctr-lifecycle PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}/build/macos/bin/Release")
    add_executable(ctr-prepare src/prepare_main.cpp)
    target_compile_options(ctr-prepare PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ctr-prepare PRIVATE ctr-runtime)
    set_target_properties(ctr-prepare PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}/build/macos/bin/Release")
endif()

# Native Mac frontend shares the runtime and Aseprite resources with iOS.
if(CTR_BUILD_FRONTEND AND NOT CTR_PLATFORM MATCHES "^ios")
    add_executable(Bergamont MACOSX_BUNDLE ios/mac_main.mm)
    target_compile_options(Bergamont PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
    target_link_libraries(Bergamont PRIVATE ctr-metal-presenter "-framework AppKit" "-framework AVFoundation" "-framework UniformTypeIdentifiers" "-framework IOKit")
    set_target_properties(Bergamont PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/ios/MacInfo.plist" BUILD_RPATH "@executable_path/../Frameworks")
    file(GLOB BG_MAC_ART "${CMAKE_CURRENT_SOURCE_DIR}/ios/PixelArt/exports/*.png")
    list(FILTER BG_MAC_ART EXCLUDE REGEX "-(icon-(gc|wii|wiiu|switch|ps1|ps2|psp)|control-(plus|minus|cross|circle|square|triangle|z|l1|l2|l3|r1|r2|r3))-[0-9]+\\.png$|-sheet\\.png$")
    target_sources(Bergamont PRIVATE ${BG_MAC_ART})
    set_source_files_properties(${BG_MAC_ART} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/PixelArt")
    add_custom_command(TARGET Bergamont POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:Bergamont>/Contents/Frameworks"
        COMMAND ${CMAKE_COMMAND} -E copy "${CORE_PATH}" "$<TARGET_BUNDLE_DIR:Bergamont>/Contents/Frameworks/azahar_libretro.dylib")
endif()

if(TARGET CTRRuntime)
 include(cmake/BergamontHandheld.cmake)
 bergamont_attach_handhelds(CTRRuntime "${CTR_PLATFORM}")
endif()
